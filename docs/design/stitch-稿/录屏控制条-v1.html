import { ScreenStreamCapture, AudioMixer } from '@rubick/core';

const targetBounds = { width: 1280, height: 720, fps: 60 };

export async function initializeCapturePipeline() {

const session = await ScreenStreamCapture.lockSelection(targetBounds);

session.attachAudioSource([AudioMixer.Microphone, AudioMixer.SystemInternal]);

session.enableH265HardwareEncoder({ bitrate: 18000 });

console.info("Rubick Recording Buffer Ready: 60 FPS");

}
